// Open the actual native ESP32-S3 USB CDC port (GPIO19/20).

using System.Text;

var port = new SerialPort(portName)
{
    BaudRate = 115200, // CDC line coding only. Link is USB Full Speed, NOT UART.
    DataBits = 8,
    Parity = Parity.None,
    StopBits = StopBits.One,
    Handshake = Handshake.None,
    DtrEnable = true, // .NET default=false; important for CDC operator bool!
    RtsEnable = false,
    ReadTimeout = 10000,
    WriteTimeout = 20000,
    NewLine = "\n",
    Encoding = Encoding.UTF8
};
port.Open();
